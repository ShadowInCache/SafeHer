# Backend + test dependencies only.
#
# requirements.txt mixes three concerns in one file: this backend, the
# Dockerised event processor, and the ML training pipeline (torch, ultralytics,
# librosa and friends -- several gigabytes that nothing under fastapi_app/ or
# tests/ imports). Installing all of it just to run the API or CI is wasteful
# and slow.
#
# Versions are pinned identically to requirements.txt; this is a subset, not a
# fork. Keep them in sync when bumping.
#
# Derived from the actual import graph:
#   grep -rhoE '^(import|from) [a-z_]+' fastapi_app/ tests/
# which yields: asyncio_mqtt, cloudinary, fastapi, httpx, jose, passlib,
# pydantic, pydantic_settings, redis, sqlalchemy (+ stdlib).

# --- web framework ---
fastapi==0.110.0
uvicorn[standard]==0.27.1
pydantic-settings==2.15.0
python-multipart==0.0.9
# Required by pydantic's EmailStr, which schemas.py uses throughout. Without it
# `import fastapi_app.main` raises ImportError at schema definition time. This
# is missing from requirements.txt too -- the dev environment only works
# because something else pulled it in transitively.
email-validator==2.1.1

# Verifies Firebase ID tokens in services/firebase_auth.py.
google-auth==2.38.0

redis[hiredis]==5.0.1
# Module name is `asyncio_mqtt`. The project was renamed upstream to `aiomqtt`,
# which installs a DIFFERENTLY named module -- swapping the two breaks
# `from asyncio_mqtt import ...` in mqtt_service.py.
asyncio-mqtt==0.13.0

# which paho-mqtt 2.x renamed to _WebsocketWrapper -- installing 2.x makes
# `import fastapi_app.main` fail with AttributeError. requirements.txt pins this
# already; the pin has to be repeated here or pip resolves 2.x transitively.
paho-mqtt==1.6.1
httpx==0.27.0
python-dotenv==1.0.0

requests==2.31.0

# --- weapon detection, web fallback only ---
#
# On Android the detector runs on the phone and the server never sees a frame.
# These exist for web clients, which have no on-device runtime.
#
# Optional by design. `services/weapon_detector.py` imports both lazily and
# reports the modality as unavailable if either is missing, so a deploy that
# omits them starts and serves normally -- web simply gets no weapon signal,
# reported as absent rather than as a zero score.
#
# `-headless` rather than plain opencv: the server has no display, and the GUI
# build fails to import without X libraries present.
# Both must be binary-compatible with numpy 2, which is what pip resolves
# here. onnxruntime 1.17 and opencv 4.8 were built against numpy 1.x and fail
# at import with "numpy.core.multiarray failed to import" -- a message that
# names neither package.
#
# Do NOT install this file into the same environment as requirements.txt: that
# one pins plain `opencv-python`, and the two opencv distributions overwrite
# each other's `cv2` module, leaving an import that half-works.
onnxruntime==1.29.0
opencv-python-headless==4.14.0.94
